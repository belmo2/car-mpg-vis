function showModels(makeObj) { d3.select("#vis").html(""); // Clear d3.select(".scene-nav").style("display", "none"); // Hide navigation in drill-down const w2 = 1200, h2 = 500, marginTop2 = 50, marginBottom2 = 130, marginLeft2 = 140, marginRight2 = 140; const svg = d3.select("#vis").append("svg") .attr("width", w2).attr("height", h2); // Use a unique label for each model: Model name + engine + index let models = makeObj.models.map((d, i) => ({ model: d.Model, mpg: ((+d.AverageCityMPG + +d.AverageHighwayMPG) / 2), label: d.Model + " (" + d.EngineCylinders + " cyl, " + i + ")" })); models = models.sort((a, b) => d3.descending(a.mpg, b.mpg)); const x = d3.scaleBand() .domain(models.map(d => d.label)) .range([marginLeft2, w2 - marginRight2]) .padding(0.2); const y = d3.scaleLinear() .domain([0, d3.max(models, d => d.mpg)]) .range([h2 - marginBottom2, marginTop2]); svg.selectAll("rect") .data(models) .join("rect") .attr("x", d => x(d.label)) .attr("y", d => y(d.mpg)) .attr("width", x.bandwidth()) .attr("height", d => y(0) - y(d.mpg)) .attr("fill", "teal"); svg.append("g") .attr("class", "x-axis-labels") .attr("transform", `translate(0,${h2 - marginBottom2})`) .call(d3.axisBottom(x).tickFormat(d => d).tickSizeOuter(0)) .selectAll("text") .attr("transform", "rotate(-65)") .attr("dx", "-0.8em") .attr("dy", "0.15em") .style("text-anchor", "end") .style("font-size", "12px"); svg.append("g") .attr("transform", `translate(${marginLeft2},0)`) .call(d3.axisLeft(y)); svg.append("text") .attr("x", w2 / 2) .attr("y", marginTop2 / 2) .attr("text-anchor", "middle") .attr("font-size", "18px") .text(`Models for ${makeObj.make} (Avg MPG)`); svg.append("text") .attr("x", w2 / 2) .attr("y", marginTop2 / 2 + 22) .attr("text-anchor", "middle") .attr("font-size", "14px") .attr("fill", "#444") .text("Each bar shows the average MPG for a model."); // Show MPG on top of each bar svg.selectAll("text.mpg-label") .data(models) .join("text") .attr("class", "mpg-label") .attr("x", d => x(d.label) + x.bandwidth() / 2) .attr("y", d => y(d.mpg) - 8) .attr("text-anchor", "middle") .text(d => d.mpg ? d.mpg.toFixed(1) : ""); // Add back button d3.select("#vis").append("button") .text("Back to Makes") .on("click", () => { updateScene(); d3.select(".scene-nav").style("display", null); // Show navigation again }); }
